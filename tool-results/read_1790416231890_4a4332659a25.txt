    31→/** Ticking studio clock (Asia/Calcutta) — hydration-safe, renders placeholder before mount. */
    32→function LiveClock() {
    33→  const [time, setTime] = useState<string | null>(null);
    34→
    35→  useEffect(() => {
    36→    const fmt = new Intl.DateTimeFormat("en-GB", {
    37→      timeZone: "Asia/Calcutta",
    38→      hour: "2-digit",
    39→      minute: "2-digit",
    40→      second: "2-digit",
    41→      hour12: false,
    42→    });
    43→    const tick = () => setTime(fmt.format(new Date()));
    44→    tick();
    45→    const t = setInterval(tick, 1000);
    46→    return () => clearInterval(t);
    47→  }, []);
    48→
    49→  return (
    50→    <span
    51→      className="inline-flex items-center gap-1.5 tabular-nums"
    52→      title="Studio local time (IST)"
    53→      suppressHydrationWarning
    54→    >
    55→      <span className="size-1.5 rounded-full bg-ibm-cyan animate-pulse-dot" aria-hidden="true" />
    56→      {time ? `Studio time ${time} IST` : "Studio time — IST"}
    57→    </span>
    58→  );
    59→}

    65→function ServicesDropdown({ active }: { active: string }) {
    66→  const [open, setOpen] = useState(false);
    67→  const wrapRef = useRef<HTMLDivElement | null>(null);
    68→  const hoverTimer = useRef<number | null>(null);
    69→  const triggerRef = useRef<HTMLButtonElement | null>(null);
    70→
    71→  const clearTimer = () => {
    72→    if (hoverTimer.current !== null) {
    73→      window.clearTimeout(hoverTimer.current);
    74→      hoverTimer.current = null;
    75→    }
    76→  };
    77→
    78→  const openWithIntent = () => {
    79→    clearTimer();
    80→    hoverTimer.current = window.setTimeout(() => setOpen(true), 90);
    81→  };
    82→
    83→  const closeWithIntent = () => {
    84→    clearTimer();
    85→    hoverTimer.current = window.setTimeout(() => setOpen(false), 260);
    86→  };
    87→
    88→  /* Escape closes and returns focus to the trigger; focus leaving the
    89→     widget (tab through the panel) also closes it. */
    90→  useEffect(() => {
    91→    if (!open) return;
    92→    const onKey = (e: KeyboardEvent) => {
    93→      if (e.key === "Escape") {
    94→        setOpen(false);
    95→        triggerRef.current?.focus();
    96→      }
    97→    };
    98→    window.addEventListener("keydown", onKey);
    99→    return () => window.removeEventListener("keydown", onKey);
   100→  }, [open]);
   101→
   102→  const panelPick = (fn: () => void) => () => {
   103→    setOpen(false);
   104→    clearTimer();
   105→    fn();
   106→  };
   107→
   108→  const isActive = active === "#services";
   109→
   110→  return (
   111→    <div
   112→      ref={wrapRef}
   113→      onMouseEnter={openWithIntent}
   114→      onMouseLeave={closeWithIntent}
   115→      onBlur={(e) => {
   116→        if (!wrapRef.current?.contains(e.relatedTarget as Node)) {
   117→          setOpen(false);
   118→          clearTimer();
   119→        }
   120→      }}
   121→    >
   122→      <button
   123→        ref={triggerRef}
   124→        type="button"
   125→        onClick={() => setOpen((v) => !v)}
   126→        onMouseEnter={openWithIntent}
   127→        aria-expanded={open}
   128→        aria-haspopup="true"
   129→        aria-controls="services-menu"
   130→        aria-current={isActive ? "true" : undefined}
   131→        className={cn(
   132→          "relative inline-flex items-center gap-1.5 font-mono text-[11px] xl:text-[13px] transition-colors focus-carbon",
   133→          isActive || open ? "text-foreground" : "text-muted-foreground hover:text-foreground"
   134→        )}
   135→      >
   136→        <Layers className="size-3.5" strokeWidth={1.5} aria-hidden="true" />

   138→        <ChevronDown
   139→          className={cn("size-3 transition-transform duration-300", open && "rotate-180")}
   140→          strokeWidth={1.5}
   141→          aria-hidden="true"
   142→        />
   143→        {isActive && (
   144→          <motion.span
   145→            layoutId="nav-underline"
   146→            className="absolute -bottom-[7px] left-0 h-[2px] w-full bg-gradient-to-r from-ibm-blue to-ibm-cyan"
   147→            transition={{ type: "spring", stiffness: 420, damping: 34 }}

   155→      <div
   156→        id="services-menu"

   158→          "absolute inset-x-0 top-full z-50 transition-all duration-200",
   159→          open
   160→            ? "pointer-events-auto translate-y-0 opacity-100"
   161→            : "pointer-events-none -translate-y-1 opacity-0"
   162→        )}
