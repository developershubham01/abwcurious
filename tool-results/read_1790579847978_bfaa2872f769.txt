   597→/* deterministic 9x9 pseudo-QR matrix */
   598→const QR_MATRIX: boolean[] = (() => {
   599→  const size = 9;
   600→  const cells: boolean[] = [];
   601→  for (let i = 0; i < size * size; i++) {
   602→    const x = i % size;
   603→    const y = Math.floor(i / size);
   604→    const finder =
   605→      (x < 3 && y < 3) || (x > 5 && y < 3) || (x < 3 && y > 5)
   606→        ? !((x === 1 && y === 1) || (x === 7 && y === 1) || (x === 1 && y === 7))
   607→        : (x * 7 + y * 13 + ((x * y) % 5)) % 3 !== 0;
   608→    cells.push(finder);
   609→  }
   610→  return cells;
   611→})();
