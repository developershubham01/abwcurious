    67→function timeAgo(iso: string) {
    68→  const diff = Date.now() - new Date(iso).getTime();
    69→  const mins = Math.floor(diff / 60000);
    70→  if (mins < 1) return "just now";
    71→  if (mins < 60) return `${mins}m ago`;
    72→  const hours = Math.floor(mins / 60);
    73→  if (hours < 24) return `${hours}h ago`;
    74→  const days = Math.floor(hours / 24);
    75→  return `${days}d ago`;
    76→}

    78→/* Carbon-style eased count-up for stat tiles */
    79→function useCountUp(target: number | undefined, active: boolean) {
    80→  const [display, setDisplay] = useState(0);
    81→  const fromRef = useRef(0);
    82→  useEffect(() => {
    83→    if (!active || typeof target !== "number") return;
    84→    const from = fromRef.current;
    85→    const duration = 550;
    86→    const t0 = performance.now();
    87→    let raf = 0;
    88→    const tick = (t: number) => {
    89→      const p = Math.min(1, (t - t0) / duration);
    90→      const eased = 1 - Math.pow(1 - p, 3);
    91→      setDisplay(Math.round(from + (target - from) * eased));
    92→      if (p < 1) {
    93→        raf = requestAnimationFrame(tick);
    94→      } else {
    95→        fromRef.current = target;
    96→      }
    97→    };
    98→    raf = requestAnimationFrame(tick);
    99→    return () => cancelAnimationFrame(raf);
   100→  }, [target, active]);
   101→  return display;
   102→}
