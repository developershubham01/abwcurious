   601→   601→  for (let i = 0; i < size * size; i++) {
   602→   602→    const x = i % size;
   603→   603→    const y = Math.floor(i / size);

   605→   605→      (x < 3 && y < 3) || (x > 5 && y < 3) || (x < 3 && y > 5)

   607→   607→        : (x * 7 + y * 13 + ((x * y) % 5)) % 3 !== 0;
   608→   608→    cells.push(finder);
   609→   609→  }
