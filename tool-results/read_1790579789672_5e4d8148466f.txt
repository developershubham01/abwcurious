   151→/** 1 Hz ticker shared by countdowns — external store keeps the hooks rule happy. */
   152→function subscribeTick(onStoreChange: () => void) {
   153→  const t = window.setInterval(onStoreChange, 1000);
   154→  return () => window.clearInterval(t);
   155→}
   156→
   157→/** Live countdown to the event date — "–" until the store ticks (hydration-safe). */
   158→function Countdown({ dateStr }: { dateStr: string }) {
   159→  const target = useMemo(() => new Date(dateStr).getTime(), [dateStr]);
   160→  const now = useSyncExternalStore(
   161→    subscribeTick,
   162→    () => Math.floor(Date.now() / 1000),
   163→    () => null
   164→  );
   165→
   166→  const valid = Number.isFinite(target);
   167→  const diff = valid && now !== null ? Math.max(0, target - now * 1000) : null;
   168→  const parts =
   169→    diff === null
   170→      ? null
   171→      : {
   172→          days: Math.floor(diff / 86_400_000),
   173→          hours: Math.floor((diff % 86_400_000) / 3_600_000),
   174→          minutes: Math.floor((diff % 3_600_000) / 60_000),
   175→          seconds: Math.floor((diff % 60_000) / 1000),
   176→        };
   177→
   178→  const cells = [
   179→    { label: "Days", value: parts?.days },
   180→    { label: "Hours", value: parts?.hours },
   181→    { label: "Min", value: parts?.minutes },
   182→    { label: "Sec", value: parts?.seconds },
   183→  ];
   184→
   185→  return (
   186→    <div className="flex items-center gap-2" role="timer" aria-label="Time until the event starts">
   187→      {cells.map((c, i) => (
   188→        <div key={c.label} className="flex items-center gap-2">
   189→          <div className="min-w-[3.4rem] rounded-xl border border-ibm-blue/15 bg-white px-2 py-1.5 text-center shadow-[0_6px_16px_-10px_rgba(15,98,254,0.45)]">
   190→            <span className="block font-mono text-base font-medium tabular-nums leading-none text-ibm-blue-active">
   191→              {c.value === null || c.value === undefined ? "–" : String(c.value).padStart(2, "0")}
   192→            </span>
   193→            <span className="mt-1 block font-mono text-[8px] uppercase tracking-[0.16em] text-ink/45">
   194→              {c.label}
   195→            </span>
   196→          </div>
   197→          {i < cells.length - 1 && (
   198→            <span className="font-mono text-sm text-ibm-blue/40" aria-hidden="true">
   199→              :
   200→            </span>
   201→          )}
   202→        </div>
   203→      ))}
   204→    </div>
   205→  );
   206→}

   208→/** Build a RFC 5545 .ics file for the event (all-day-ish 8h hold, UTC). */
   209→function buildICS(event: CompanyEvent): string {
   210→  const start = new Date(event.date);
   211→  const end = new Date(start.getTime() + 8 * 3_600_000);
   212→  const fmt = (d: Date) => d.toISOString().replace(/[-:]/g, "").split(".")[0] + "Z";
   213→  const esc = (s: string) => s.replace(/\\/g, "\\\\").replace(/;/g, "\\;").replace(/,/g, "\\,").replace(/\n/g, "\\n");
   214→  return [
   215→    "BEGIN:VCALENDAR",
   216→    "VERSION:2.0",
   217→    "PRODID:-//ABWcurious//Company Events//EN",
   218→    "CALSCALE:GREGORIAN",
   219→    "BEGIN:VEVENT",
   220→    `UID:${event.id}@abwcurious`,
   221→    `DTSTAMP:${fmt(new Date())}`,
   222→    `DTSTART:${fmt(start)}`,
   223→    `DTEND:${fmt(end)}`,
   224→    `SUMMARY:${esc(event.name)}`,
   225→    `LOCATION:${esc(event.location)}`,
   226→    `DESCRIPTION:${esc(event.description)}`,
   227→    `URL:${typeof window !== "undefined" ? window.location.origin + window.location.pathname : ""}`,
   228→    "END:VEVENT",
   229→    "END:VCALENDAR",
   230→  ].join("\r\n");
   231→}

   233→function AddToCalendarButton({ event }: { event: CompanyEvent }) {
   234→  const [done, setDone] = useState(false);
   235→
   236→  function download() {
   237→    try {
   238→      const blob = new Blob([buildICS(event)], { type: "text/calendar;charset=utf-8" });
   239→      const url = URL.createObjectURL(blob);
   240→      const a = document.createElement("a");
   241→      a.href = url;
   242→      a.download = `${event.id}.ics`;
   243→      document.body.appendChild(a);
   244→      a.click();
   245→      a.remove();
   246→      URL.revokeObjectURL(url);
   247→      setDone(true);
   248→      window.setTimeout(() => setDone(false), 2600);
   249→    } catch {
   250→      /* download blocked — silent, button stays */
   251→    }
   252→  }
   253→
   254→  return (
   255→    <button
   256→      type="button"
   257→      onClick={download}
   258→      className={cn(
   259→        "focus-carbon inline-flex h-10 items-center gap-2 rounded-full border border-ibm-blue/30 bg-white px-4",
   260→        "font-mono text-[10px] uppercase tracking-[0.14em] text-ibm-bright transition-colors hover:bg-ibm-blue hover:text-white",
   261→        done && "border-ibm-success/40 text-ibm-success hover:bg-ibm-success hover:text-white"
   262→      )}
   263→    >
   264→      {done ? (
   265→        <CheckCircle2 className="size-3.5" strokeWidth={1.75} aria-hidden="true" />
   266→      ) : (
   267→        <CalendarPlus className="size-3.5" strokeWidth={1.75} aria-hidden="true" />
   268→      )}
   269→      {done ? "Saved" : "Add to calendar"}
   270→    </button>
   271→  );
   272→}

   274→/** Email waitlist for upcoming events → /api/newsletter (source: "event:<id>"). */
   275→function WaitlistForm({ event }: { event: CompanyEvent }) {
   276→  const { toast } = useToast();
   277→  const [email, setEmail] = useState("");
   278→  const [state, setState] = useState<"idle" | "loading" | "done">("idle");
   279→
   280→  async function submit(e: React.FormEvent<HTMLFormElement>) {
   281→    e.preventDefault();
   282→    if (!email || state === "loading") return;
   283→    setState("loading");
   284→    try {
   285→      const res = await fetch("/api/newsletter", {
   286→        method: "POST",
   287→        headers: { "Content-Type": "application/json" },
   288→        body: JSON.stringify({ email, source: `event:${event.id}`.slice(0, 60) }),
   289→      });
   290→      const json = await res.json();
   291→      if (!res.ok) throw new Error(json.error || "Could not sign you up");
   292→      setState("done");
   293→      toast({
   294→        title: json.alreadySubscribed ? "Already on the list" : "You're on the list",
   295→        description: json.alreadySubscribed
   296→          ? "We'll email you when this event opens up."
   297→          : `We'll remind you before ${event.name}.`,
   298→      });
   299→    } catch (err) {
   300→      setState("idle");
   301→      toast({
   302→        title: "Could not sign you up",
   303→        description: err instanceof Error ? err.message : "Please try again.",
   304→        variant: "destructive",
   305→      });
   306→    }
   307→  }
   308→
   309→  if (state === "done") {
   310→    return (
   311→      <div className="flex items-center gap-2.5 rounded-2xl border border-ibm-success/30 bg-ibm-success/[0.08] px-4 py-3">
   312→        <CheckCircle2 className="size-4 shrink-0 text-ibm-success" strokeWidth={1.75} aria-hidden="true" />
   313→        <p className="font-mono text-[11px] uppercase tracking-[0.1em] text-ink/70">
   314→          Reminder set — see you there
   315→        </p>
   316→      </div>
   317→    );
   318→  }
   319→
   320→  return (
   321→    <form
   322→      onSubmit={submit}
   323→      className="flex overflow-hidden rounded-2xl border border-ibm-blue/20 bg-white shadow-[0_8px_24px_-14px_rgba(15,98,254,0.45)] transition-colors focus-within:border-ibm-blue/60"
   324→    >
   325→      <label htmlFor={`waitlist-${event.id}`} className="sr-only">
   326→        Email for event reminders
   327→      </label>
   328→      <input
   329→        id={`waitlist-${event.id}`}
   330→        type="email"
   331→        required
   332→        value={email}
   333→        onChange={(e) => setEmail(e.target.value)}
   334→        placeholder="you@company.com"
   335→        autoComplete="email"
   336→        className="h-10 w-full min-w-0 bg-transparent px-4 font-mono text-xs text-ink placeholder:text-ink/35 focus:outline-none"
   337→      />
   338→      <button
   339→        type="submit"
   340→        disabled={state === "loading"}
   341→        className="inline-flex h-10 shrink-0 items-center gap-1.5 bg-ibm-blue px-4 font-mono text-[10px] uppercase tracking-[0.14em] text-white transition-colors hover:bg-ibm-blue-hover focus-carbon disabled:opacity-60"
   342→      >
   343→        {state === "loading" ? (
   344→          <LoaderCircle className="size-3.5 animate-spin" aria-hidden="true" />
   345→        ) : (
   346→          <BellRing className="size-3.5" strokeWidth={1.75} aria-hidden="true" />
   347→        )}
   348→        Notify me
   349→      </button>
   350→    </form>
   351→  );
   352→}
