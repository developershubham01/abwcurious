    31→/** Ticking studio clock (Asia/Calcutta) — hydration-safe, renders placeholder before mount. */
    32→function LiveClock() {
    33→  const [time, setTime] = useState<string | null>(null);
    34→
    35→  useEffect(() => {
    36→    const fmt = new Intl.DateTimeFormat("en-GB", {
    37→      timeZone: "Asia/Calcutta",
    38→      hour: "2-digit",
    39→      minute: "2-digit",
    40→      second: "2-digit",
    41→      hour12: false,
    42→    });
    43→    const tick = () => setTime(fmt.format(new Date()));
    44→    tick();
    45→    const t = setInterval(tick, 1000);
    46→    return () => clearInterval(t);
    47→  }, []);
    48→
    49→  return (
    50→    <span
    51→      className="inline-flex items-center gap-1.5 tabular-nums"
    52→      title="Studio local time (IST)"
    53→      suppressHydrationWarning
    54→    >
    55→      <span className="size-1.5 rounded-full bg-ibm-cyan animate-pulse-dot" aria-hidden="true" />
    56→      {time ? `Studio time ${time} IST` : "Studio time — IST"}
    57→    </span>
    58→  );
    59→}

    65→function ServicesDropdown({ active, variant = "site" }: { active: string; variant?: "site" | "view" }) {

    70→  const underlineId = variant === "view" ? "nav-underline-view" : "nav-underline";
    71→  const menuId = variant === "view" ? "services-menu-view" : "services-menu";
    72→
    73→  const clearTimer = () => {
    74→    if (hoverTimer.current !== null) {
    75→      window.clearTimeout(hoverTimer.current);
    76→      hoverTimer.current = null;
    77→    }
    78→  };
    79→
    80→  const openWithIntent = () => {
    81→    clearTimer();
    82→    hoverTimer.current = window.setTimeout(() => setOpen(true), 90);
    83→  };
    84→
    85→  const closeWithIntent = () => {
    86→    clearTimer();
    87→    hoverTimer.current = window.setTimeout(() => setOpen(false), 260);
    88→  };
    89→
    90→  /* Escape closes and returns focus to the trigger; focus leaving the
    91→     widget (tab through the panel) also closes it. */
    92→  useEffect(() => {
    93→    if (!open) return;
    94→    const onKey = (e: KeyboardEvent) => {
    95→      if (e.key === "Escape") {
    96→        setOpen(false);
    97→        triggerRef.current?.focus();
    98→      }
    99→    };
   100→    window.addEventListener("keydown", onKey);
   101→    return () => window.removeEventListener("keydown", onKey);
   102→  }, [open]);
   103→
   104→  const panelPick = (fn: () => void) => () => {
   105→    setOpen(false);
   106→    clearTimer();
   107→    fn();
   108→  };

   110→  const isActive = active === "#services";

   112→  return (
   113→    <div
   114→      ref={wrapRef}
   115→      onMouseEnter={openWithIntent}
   116→      onMouseLeave={closeWithIntent}
   117→      onBlur={(e) => {
   118→        if (!wrapRef.current?.contains(e.relatedTarget as Node)) {
   119→          setOpen(false);
   120→          clearTimer();
   121→        }
   122→      }}
   123→    >
   124→      <button
   125→        ref={triggerRef}
   126→        type="button"
   127→        onClick={() => setOpen((v) => !v)}

   129→        aria-expanded={open}
   130→        aria-haspopup="true"
   131→        aria-controls={menuId}
   132→        aria-current={isActive ? "true" : undefined}
   133→        className={cn(
   134→          "relative inline-flex items-center gap-1.5 font-mono text-[11px] xl:text-[13px] transition-colors focus-carbon",
   135→          isActive || open ? "text-foreground" : "text-muted-foreground hover:text-foreground"

   140→        <ChevronDown
   141→          className={cn("size-3 transition-transform duration-300", open && "rotate-180")}
   142→          strokeWidth={1.5}
   143→          aria-hidden="true"
   144→        />
   145→        {isActive && (
   146→          <motion.span
   147→            layoutId={underlineId}
   148→            className="absolute -bottom-[7px] left-0 h-[2px] w-full bg-gradient-to-r from-ibm-blue to-ibm-cyan"
